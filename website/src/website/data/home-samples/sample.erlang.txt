%% A simple Erlang module
-module(sample).
-export([factorial/1, greet/1]).

%% Recursive factorial
factorial(0) ->
    1;
factorial(N) when N > 0 ->
    N * factorial(N - 1).

%% Greet a person by name
greet(Name) ->
    Message = "Hello, " ++ Name ++ "!",
    io:format("~s~n", [Message]),
    Message.
